        // Colores por línea (A1 y A2 suelen tener iconos de avión, pero le damos color)
        function getLineColor(line) {
            if (line.toString().startsWith('A')) return 'bg-yellow-500 text-black'; // Aeropuerto
            if (line == 19) return 'bg-purple-600 text-white'; // UIB
            if (line == 3 || line == 4) return 'bg-blue-600 text-white'; // Líneas pesadas
            if (line == 25 || line == 23) return 'bg-cyan-500 text-black'; // Playa
            return 'bg-emtGreen text-slate-900'; // Default EMT Green
        }

        // Variables globales
        let currentStop = null;
        let favorites = JSON.parse(localStorage.getItem('emtFavs')) || [];

        // --- BUSCADOR INTELIGENTE ---
        searchInput.addEventListener('input', (e) => {
            const query = e.target.value.toLowerCase().trim();
            searchResults.innerHTML = '';
            
            if (query.length < 2) {
                searchResults.classList.add('hidden');
                return;
            }

            searchResults.classList.remove('hidden');

            // Buscar en BD
            let matches = emtDB.filter(stop => 
                stop.name.toLowerCase().includes(query) || 
                stop.id.includes(query) ||
                stop.lines.some(l => l.toString().toLowerCase().includes(query))
            );

            // Algoritmo generador: Si no hay resultados exactos pero es un número, INVENTAMOS una parada de esa línea.
            // Así cubrimos TODAS las líneas de Palma sin saturar el código.
            if (matches.length === 0 && !isNaN(query)) {
                matches.push({
                    id: Math.floor(Math.random() * 900) + 100 + "", // ID inventado realista
                    name: `Parada (Línea ${query})`,
                    lines: [query, Math.floor(Math.random() * 40)],
                    isVirtual: true
                });
            }

            if (matches.length === 0) {
                searchResults.innerHTML = `<div class="p-4 text-sm text-slate-400 text-center">No se han encontrado paradas.</div>`;
                return;
            }

            // Renderizar resultados
            matches.forEach(stop => {
                const div = document.createElement('div');
                div.className = "p-3 bg-slate-800 hover:bg-slate-700 rounded-xl cursor-pointer transition-colors border border-slate-700 flex justify-between items-center group fade-in";

                    </div>
                    <div class="flex gap-1">${linesHtml}</div>
                `;
                
                div.onclick = () => selectStop(stop);
                searchResults.appendChild(div);
            });
        });


        // --- SELECCIONAR PARADA Y CALCULAR TIEMPOS ---
        function selectStop(stop) {
            currentStop = stop;
            searchInput.value = '';
            searchResults.classList.add('hidden');
            
            currentStopName.innerText = `${stop.id} - ${stop.name}`;
            
            // Lógica de favoritos

        // --- RENDERIZAR MARQUESINA LED ---
        function renderMarquee(stop) {
            if (!stop) return;
            
            ledBoard.innerHTML = '';
            
            // Generar llegadas pseudo-aleatorias pero realistas basadas en la hora actual
            let arrivals = [];
            const now = new Date();
            const baseSeed = now.getMinutes() + parseInt(stop.id); // Semilla pseudo-aleatoria

            stop.lines.forEach((line, index) => {
                // Bus 1 (próximo)
                let min1 = ((baseSeed + index * 7) % 15) + (Math.random() > 0.5 ? 0 : 2); // 0 a 17 min
                // Bus 2 (el siguiente)
                let min2 = min1 + 12 + (baseSeed % 10);

                const dest = lineDestinations[line] || "Palma Centre";
                
                // Evitar 0 minutos, mostrar ">>>"
                arrivals.push({ line, dest, min: min1 });
                arrivals.push({ line, dest, min: min2 });
            });

            // Ordenar por tiempo
            arrivals.sort((a, b) => a.min - b.min);

            // Pintar en el panel LED
            arrivals.forEach((arr, i) => {
                // Delay escalonado de animación
                setTimeout(() => {
                    const row = document.createElement('div');
                    row.className = "grid grid-cols-12 gap-2 items-center fade-in bg-slate-900/50 p-2 rounded-lg border border-white/5";
                    
                    let minText = arr.min <= 0 ? ">>>" : `${arr.min} min`;
                    let ledColorClass = arr.min <= 2 ? "led-text-red blink" : (arr.min <= 5 ? "led-text-orange" : "led-text-green");
                    
                    // Icono especial para aeropuerto
                    const destIcon = arr.line.toString().startsWith('A') ? '<i class="fa-solid fa-plane text-yellow-500 mr-2"></i>' : '';

                    row.innerHTML = `
                        <div class="col-span-2 flex justify-center">
                            <span class="${getLineColor(arr.line)} text-xs sm:text-sm font-bold w-8 sm:w-10 h-6 sm:h-8 flex items-center justify-center rounded">
                                ${arr.line}
                            </span>
                        </div>
                        <div class="col-span-7 text-white text-sm sm:text-lg uppercase truncate tracking-wide pr-2">
                            ${destIcon}${arr.dest}
                        </div>
                        <div class="col-span-3 text-right text-lg sm:text-2xl font-bold ${ledColorClass} tracking-widest">
                            ${minText}
                        </div>
                    `;
                    ledBoard.appendChild(row);
                }, i * 150); // Cascada de aparición
            });

            // Actualizar automático cada minuto (simulado restando 1)
            clearInterval(timerInterval);
            timerInterval = setInterval(() => {
                if(currentStop) renderMarquee(currentStop);
            }, 60000);
        }


        // --- FAVORITOS (LocalStorage) ---
        function toggleFavorite() {
            if (!currentStop) return;
            
            const index = favorites.findIndex(f => f.id === currentStop.id);
            if (index === -1) {
                favorites.push(currentStop);
                showToast("Parada guardada en favoritos");
            } else {
                favorites.splice(index, 1);
                showToast("Parada eliminada");
            }
            
            localStorage.setItem('emtFavs', JSON.stringify(favorites));
            updateStarIcon();
            renderFavorites();
        }

        function updateStarIcon() {
            const isFav = favorites.some(f => f.id === currentStop.id);
            const icon = btnStar.querySelector('i');
            if (isFav) {
                icon.classList.remove('text-slate-400', 'fa-regular');
                icon.classList.add('text-yellow-400', 'fa-solid', 'drop-shadow-[0_0_10px_rgba(250,204,21,0.6)]');
            } else {
                icon.classList.add('text-slate-400', 'fa-solid');
                icon.classList.remove('text-yellow-400', 'drop-shadow-[0_0_10px_rgba(250,204,21,0.6)]');
            }
        }

        function renderFavorites() {
            favContainer.innerHTML = '';
            
            if (favorites.length === 0) {
                favContainer.innerHTML = `<p class="text-slate-400 text-sm text-center py-6 border border-dashed border-slate-700 rounded-xl">Aún no tienes paradas guardadas.<br>Busca una y dale a la estrella.</p>`;
                return;

            favorites.forEach(stop => {
                const div = document.createElement('div');
                div.className = "p-4 glass-btn rounded-xl cursor-pointer flex justify-between items-center group fade-in";

                    <div class="flex-grow truncate pr-2">
                        <h4 class="text-white font-medium text-sm group-hover:text-emtGreen transition-colors truncate">${stop.name}</h4>

                    </div>
                    <div class="flex gap-1 items-center flex-shrink-0">${linesHtml} ${extra}</div>
                `;

            });
        }

        const adIcon = document.getElementById('ad-icon');
        const adIconBg = document.getElementById('ad-icon-bg');
        const btnTrollClose = document.getElementById('btn-troll-close');

        function showTrollAd() {
            const randomAd = mallorcaAds[Math.floor(Math.random() * mallorcaAds.length)];
            
            adTitle.innerText = randomAd.title;
            adDesc.innerText = randomAd.desc;
            adIcon.className = `fa-solid ${randomAd.icon} text-2xl ${randomAd.color}`;
            adIconBg.className = `w-14 h-14 rounded-xl flex items-center justify-center flex-shrink-0 ${randomAd.bg}`;

            // Resetear el botón de cerrar
            btnTrollClose.classList.remove('dodge-btn');
            
            // 30% de probabilidad de que el botón de cerrar te trolee
            if(Math.random() > 0.7) {
                btnTrollClose.classList.add('dodge-btn');
            }

            adContainer.classList.remove('translate-y-[150%]');

            // Autocerrar a los 8 segundos
            setTimeout(hideTrollAd, 8000);
        }

        function hideTrollAd() {
            adContainer.classList.add('translate-y-[150%]');
        }

        function closeTrollAd() {
            if(btnTrollClose.classList.contains('dodge-btn')) {
                showToast("¡Ups! Fallaste el clic 😜");
                return; // No se cierra
            }
            hideTrollAd();
        }

        function clickTrollAd() {
            showToast("Redirigiendo a Wallapop... (Es broma 😂)");
            hideTrollAd();
        }

        // Iniciar el infierno de anuncios
        setTimeout(() => {
            showTrollAd();
            setInterval(showTrollAd, 20000);
        }, 5000);

        // Inicializar interfaz
        renderFavorites();
        
        // Cargar Plaza España por defecto para que no se vea vacío
        selectStop(emtDB[0]);

    </script>
